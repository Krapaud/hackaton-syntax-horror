    <script>
        const envelope = document.getElementById('envelope');
        const instruction = document.getElementById('instruction');
        const downloadBtn = document.getElementById('downloadBtn');
        let isOpened = false;
        let isUnlocked = false;

        envelope.addEventListener('click', function() {
            if (!isOpened) {
                envelope.classList.add('opened');
                instruction.textContent = 'Résolvez l\'énigme pour déverrouiller le téléchargement';
                isOpened = true;
            }
        });

        function verifyPassword() {
            const input = document.getElementById('passwordInput');
            const result = document.getElementById('result');
            const password = input.value.trim().toUpperCase();

            if (password === 'JASON') {
                result.innerHTML = '<div class="message-result success"> ACCÈS ACCORDÉ ! Le mot de passe est correct.</div>';
                isUnlocked = true;
                downloadBtn.classList.add('active');
                instruction.textContent = 'Vous pouvez maintenant télécharger le message !';
            } else {
                result.innerHTML = '<div class="message-result error"> Mauvaise réponse. Relisez les indices...</div>';
                setTimeout(() => {
                    result.innerHTML = '';
                }, 3000);
            }
        }

        document.getElementById('passwordInput').addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                verifyPassword();
            }
        });

        function downloadLetter() {
            if (!isUnlocked) {
                alert('Vous devez d\'abord résoudre l\'énigme !');
                return;
            }

            const content = `
═══════════════════════════════════════════════════
          MESSAGE DÉVERROUILLÉ  
═══════════════════════════════════════════════════

FÉLICITATIONS !

Vous avez résolu l'énigme et découvert l'identité 
du tueur légendaire de Crystal Lake.

╔═══════════════════════════════════════╗
║                                                ║
║      VOTRE MOT DE PASSE :                      ║
║                                                ║
║              FRI13TH                           ║
║                                                ║
╚═══════════════════════════════════════╝

L'ÉNIGME :
- Au bord d'un lac (Crystal Lake)
- Le jour le plus malchanceux (Vendredi 13)
- Masque de hockey blanc iconique
- Machete et immortalité
- Vengeance de la mère (Pamela Voorhees)
- Le tueur : JASON VOORHEES

 ACCÈS SÉCURISÉ CONFIRMÉ 

---------------------------------------------------
Film : Vendredi 13 (Friday the 13th)
Personnage : Jason Voorhees
Date de déverrouillage : ${new Date().toLocaleDateString('fr-FR')}
═══════════════════════════════════════════════════
            `;

            const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'message_deverrouille_JASON.txt';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            window.URL.revokeObjectURL(url);
        }
